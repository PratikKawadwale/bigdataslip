1. Create a database named IRating.
CREATE DATABASE IRating;


2. Use the created database.
USE IRating;


3. Create a table Rating with fields separated by tab (\t).
CREATE TABLE Rating (
    userid INT,
    item_id INT,
    rating FLOAT,
    timestamp BIGINT
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY '\t';


4. Load data from local storage.
LOAD DATA LOCAL INPATH '/home/user/data/Item_Rating'
INTO TABLE Rating;


5. Find the average number of ratings per Item.
SELECT item_id, AVG(rating) AS average_rating
FROM Rating
GROUP BY item_id;


6. List the Item_id of all items which have received rating greater than 4.
SELECT DISTINCT item_id
FROM Rating
WHERE rating > 4;

a. Load the file employee.csv.
employee = LOAD 'employee.csv'
USING PigStorage(',')
AS (eno:int, ename:chararray, salary:double, dept:chararray, age:int);


b. Display employee having maximum age.
grouped = GROUP employee ALL;
max_age = FOREACH grouped GENERATE MAX(employee.age) AS age;

result = JOIN employee BY age, max_age BY age;
DUMP result;


c. Dump department-wise sum of salary.
grouped_dept = GROUP employee BY dept;
salary_sum = FOREACH grouped_dept GENERATE
    group AS dept,
    SUM(employee.salary) AS total_salary;
DUMP salary_sum;


d. Dump all employees in descending order of employee name.
sorted_employee = ORDER employee BY ename DESC;
DUMP sorted_employee;


e. Display employees of Sales department.
sales_dept = FILTER employee BY dept == 'Sales';
DUMP sales_dept;
